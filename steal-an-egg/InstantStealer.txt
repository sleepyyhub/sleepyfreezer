--[[
    Luminosity // Instant Stealer  v2  --  Steal An Egg (placeId 107778070777162)
    Executor: any with gethui/getgenv (built against Delta 1.0.733.988)
    Language: Luau, client-side. Paste and execute.

    Tabs
      STEAL  auto/manual stealing, banking, filters
      EGGS   every live egg with its real values, click a row to teleport to it
      ZONES  one-tap teleport to all 10 guard areas, your plot, spawn
      INFO   what the numbers mean + the endpoints this uses

    Mechanism
      Stealing is a vanilla ProximityPrompt named CarryAreaEgg (8 studs, 1.2s hold)
      whose Triggered handler calls the RemoteFunction "Eggs: RequestAreaEggCarry"
      with (uid, firstAreaSlotKey?) and gets back (ok, denyReason). This skips the
      prompt and the hold: read the snapshot, pivot onto the egg for one frame so
      the server range check passes, invoke the carry, pivot home to bank.

      Values come from ReplicatedStorage.Directory.Assets.Directory[AssetCategory]
      (EarningRate, Rarity, VisualOdds, Egg.GrowthTime, Egg.WeightKg) and from
      Library.Modules.Mutations (ValueMulti per mutation).

local AssetDir, MutationDir

do
    local ok, mod = pcall(function() return require(RS.Directory.Assets) end)
    if ok and type(mod) == "table" then AssetDir = mod.Directory or mod end
end

do
    -- The Mutations module exposes each mutation with a ValueMulti; its exact
    -- container name moves between versions, so probe for the shape instead.
    local ok, mod = pcall(function() return require(RS.Library.Modules.Mutations) end)
    if ok and type(mod) == "table" then
        local function looksRight(t)
            if type(t) ~= "table" then return false end
            for _, v in pairs(t) do
                if type(v) == "table" and tonumber(v.ValueMulti) then return true end
            end
            return false
        end
        for _, key in ipairs({"Mutations", "All", "Directory", "List", "Types"}) do
            if looksRight(rawget(mod, key)) then MutationDir = mod[key] break end
        end
        if not MutationDir and looksRight(mod) then MutationDir = mod end
    end
end

-- Guard escape maths --------------------------------------------------------
--
-- This is why you were dying. Taking an egg wakes the area guard and it chases.
-- Each area's guard has a minimum escape speed; under it the guard reaches its
-- hit distance and takes you down. Your speed power right now is whatever
-- SpeedPowerProjection reports (base is 10, i.e. WalkSpeed ~20).

local AreaDir, GuardDir, ResolveGuardSpeed, SpeedProjection, TreadUtil

do
    local ok, mod = pcall(function() return require(RS.Directory.Areas) end)
    if ok and type(mod) == "table" then AreaDir = mod.Directory or mod end
end
do
    local ok, mod = pcall(function() return require(RS.Directory.Guards) end)
    if ok and type(mod) == "table" then GuardDir = mod.Directory or mod end
end
do
    local ok, mod = pcall(function() return require(RS.Library.Functions.ResolveGuardSpeedRequirement) end)
    if ok and type(mod) == "function" then ResolveGuardSpeed = mod end
end
do
    local ok, mod = pcall(function() return require(RS.Library.Client.SpeedPowerProjection) end)
    if ok and type(mod) == "table" then SpeedProjection = mod end
end
do
    local ok, mod = pcall(function() return require(RS.Library.Util.TreadmillUtil) end)
    if ok and type(mod) == "table" then TreadUtil = mod end
end

local function mySpeedPower()
    if SpeedProjection and SpeedProjection.GetSpeedPower then
        local ok, v = pcall(SpeedProjection.GetSpeedPower)
        if ok and tonumber(v) then return tonumber(v) end
    end
    return (TreadUtil and TreadUtil.DEFAULT_BASE_SPEED_POWER) or 10
end

local guardReqCache = {}

-- Required speed power to outrun the guard of this area, or nil if unknown.
local function areaSpeedRequirement(areaId)
    if guardReqCache[areaId] ~= nil then
        return guardReqCache[areaId] or nil
    end
    local req
    local okArea, areaCfg = pcall(function() return AreaDir and AreaDir[areaId] end)
    if okArea and type(areaCfg) == "table" and areaCfg.GuardId and GuardDir and ResolveGuardSpeed then
        for _, guardCfg in pairs(GuardDir) do
            if type(guardCfg) == "table" and guardCfg._id == areaCfg.GuardId then
                local okReq, value = pcall(ResolveGuardSpeed, guardCfg)
                if okReq and tonumber(value) then req = tonumber(value) end
                break
            end
        end
    end
    guardReqCache[areaId] = req or false
    return req
end

local function canEscape(areaId)
    local req = areaSpeedRequirement(areaId)
    if not req then return true, nil end       -- unknown, do not block
    return mySpeedPower() >= req, req
end

local function assetConfig(category)
    if not (AssetDir and category) then return nil end
    local ok, cfg = pcall(function() return AssetDir[category] end)
    return ok and cfg or nil
end

local function mutationMulti(name)
    if not (MutationDir and name) then return 1 end
    local ok, entry = pcall(function() return MutationDir[name] end)
    if ok and type(entry) == "table" and tonumber(entry.ValueMulti) then
        return tonumber(entry.ValueMulti)
    end
    return 1
end

-- 1234567 -> "1.23M"
local function short(n)
    n = tonumber(n) or 0
    local units = {
        {1e15, "Q"}, {1e12, "T"}, {1e9, "B"}, {1e6, "M"}, {1e3, "K"},
    }
    for _, u in ipairs(units) do
        if math.abs(n) >= u[1] then
            return string.format("%.2f%s", n / u[1], u[2])
        end
    end
    return string.format("%d", math.floor(n + 0.5))
end

local function hhmm(seconds)
    seconds = tonumber(seconds) or 0
    local h = math.floor(seconds / 3600)
    local m = math.floor((seconds % 3600) / 60)
    if h > 0 then return string.format("%dh%02dm", h, m) end
    return string.format("%dm", m)
end

-- Everything the UI needs to describe one egg.
local function describe(record)
    local cfg = assetConfig(record.AssetCategory)

    local info = {
        uid       = record.Uid,
        category  = record.AssetCategory,
        name      = record.AssetCategory or "?",
        rarity    = "Unknown",
        rarityNum = 0,
        color     = Color3.fromRGB(200, 210, 225),
        base      = 0,
        multi     = 1,
        value     = 0,
        odds      = nil,
        growth    = nil,
        weight    = nil,
        scale     = tonumber(record.AssetScale) or 1,
        mutations = {},
        area      = record.AreaId or "?",
        state     = record.State,
    }

    if cfg then
        info.name   = cfg.DisplayName or info.name
        info.base   = tonumber(cfg.EarningRate) or 0
        info.odds   = tonumber(cfg.VisualOdds)
        local rar = cfg.Rarity
        if type(rar) == "table" then
            info.rarity    = rar.DisplayName or info.rarity
            info.rarityNum = tonumber(rar.RarityNumber) or 0
            if typeof(rar.Color) == "Color3" then info.color = rar.Color end
        end
        local egg = cfg.Egg
        if type(egg) == "table" then
            info.growth = tonumber(egg.GrowthTime)
            info.weight = tonumber(egg.WeightKg)
        end
    end

    if type(record.Mutations) == "table" then
        for _, m in ipairs(record.Mutations) do
            local mult = mutationMulti(m)
            info.mutations[#info.mutations + 1] = {name = m, multi = mult}
            info.multi = info.multi * mult
        end
    end
    if record.BaseMutation and record.BaseMutation ~= "" then
        info.baseMutation = record.BaseMutation
    end

    info.value = info.base * info.multi
    return info
end

local Net = {}

local function endpoint(name)
    return NetFolder and NetFolder:FindFirstChild(name) or nil
end

function Net.Snapshot()
    local rf = endpoint("Eggs: RequestAreaEggSnapshot")
    if not rf then return nil end
    local ok, snap = pcall(function() return rf:InvokeServer() end)
    if ok and type(snap) == "table" and type(snap.Records) == "table" then
        return snap.Records
    end
    return nil
end

function Net.Carry(uid, slotKey)
    local rf = endpoint("Eggs: RequestAreaEggCarry")
    if not rf then return false, "endpoint missing" end
    local ok, res, reason = pcall(function() return rf:InvokeServer(uid, slotKey) end)
    if not ok then return false, tostring(res) end
    return res == true, reason
end

function Net.Drop(reason)
    local rf = endpoint("Eggs: RequestAreaEggDrop")
    if not rf then return false end
    return (pcall(function() return rf:InvokeServer(reason or "PlayerRequest") end))
end

do
    local ev = endpoint("Eggs: AreaEggCarryState")
    if ev then
        ev.OnClientEvent:Connect(function(payload)
            if type(payload) ~= "table" then return end
            STATE.carrying = payload.IsCarrying == true
            STATE.carryUid = payload.Uid
        end)
    end
end

local function isFirstAreaUid(uid) return string.find(uid, "FirstAreaEgg_", 1, true) == 1 end

local function firstAreaOwner(uid)
    local id = string.match(uid, "^FirstAreaEgg_(-?%d+)_")
    return id and tonumber(id) or nil
end

local function slotKeyFor(record)
    if isFirstAreaUid(record.Uid) then
        return string.format("%s:%s", record.AreaId, record.NestId)
    end
    return nil
end

local function rootPart()
    local char = LP.Character
    return char and char:FindFirstChild("HumanoidRootPart") or nil
end

local function eggPosition(record)
    local cf = record.BottomCFrame
    return typeof(cf) == "CFrame" and cf.Position or nil
end

local function isStealable(record)
    if record.CarrierUserId ~= nil then return false end
    if record.State ~= EGG_STATES.Slot then
        if not (record.State == EGG_STATES.Dropped and CFG.IncludeDropped) then
            return false
        end
    end
    if CFG.SkipOwnArea and isFirstAreaUid(record.Uid) and firstAreaOwner(record.Uid) == LP.UserId then
        return false
    end
    return eggPosition(record) ~= nil
end

local function collectTargets()
    local records = Net.Snapshot()
    if not records then return STATE.targets end

    local hrp = rootPart()
    local origin = hrp and hrp.Position or Vector3.zero

    local out = {}
    for _, record in ipairs(records) do
        if isStealable(record) then
            local pos = eggPosition(record)
            local info = describe(record)
            info.position = pos
            info.distance = (pos - origin).Magnitude
            info.record = record
            out[#out + 1] = info
        end
    end

    local mode = CFG.SortMode
    table.sort(out, function(a, b)
        if mode == 2 then return a.distance < b.distance end
        if mode == 3 then
            if a.rarityNum ~= b.rarityNum then return a.rarityNum > b.rarityNum end
            return a.distance < b.distance
        end
        if a.value ~= b.value then return a.value > b.value end
        return a.distance < b.distance
    end)

    STATE.available = #out
    STATE.targets = out
    return out
end

-- Rubberband control ------------------------------------------------------
--
-- Two separate things were throwing you back. One: a single PivotTo across the
-- map is a position delta no legitimate character can produce, so the server's
-- own character replication rejects it and snaps you to its last good value.
-- Stepping the trip fixes that - the server sees fast but continuous motion.
-- Two: the old bank() re-pivoted EVERY heartbeat, which fights the Humanoid's
-- physics state and produces the classic stutter even when the destination is
-- fine. Now it corrects only after real drift.

local humanoid = function()
    local char = LP.Character
    return char and char:FindFirstChildOfClass("Humanoid") or nil
end

local clipCache = {}

local function setNoClip(on)
    local char = LP.Character
    if not char then return end
    if on then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then
                clipCache[part] = true
                part.CanCollide = false
            end
        end
    else
        for part in pairs(clipCache) do
            if part.Parent then part.CanCollide = true end
        end
        table.clear(clipCache)
    end
end

local function settle(hrp)
    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
end

-- Put the Humanoid somewhere that will not argue with a hard position set, and
-- kill the fall so the landing cannot deal damage.
local function prepareMove()
    local hum = humanoid()
    if hum then
        pcall(function() hum:ChangeState(Enum.HumanoidStateType.Freefall) end)
    end
    if CFG.NoClip then setNoClip(true) end
end

local function finishMove()
    if CFG.NoClip then setNoClip(false) end
    local hum, hrp = humanoid(), rootPart()
    if hrp then settle(hrp) end
    if hum then
        pcall(function() hum:ChangeState(Enum.HumanoidStateType.GettingUp) end)
    end
end

local function pivotRaw(cframe)
    local char = LP.Character
    local hrp = rootPart()
    if not (char and hrp) then return false end
    char:PivotTo(cframe)
    settle(hrp)
    return true
end

-- The main mover. Stepped by default; set CFG.Stepped false for the old
-- one-frame jump if a given server does not care.
local function travel(target)
    local hrp = rootPart()
    if not hrp then return false end

    if not CFG.Stepped then
        prepareMove()
        local ok = pivotRaw(target)
        finishMove()
        return ok
    end

    prepareMove()

    local start = hrp.CFrame
    local delta = target.Position - start.Position
    local dist = delta.Magnitude
    local hops = math.max(1, math.ceil(dist / math.max(CFG.HopStuds, 1)))

    for i = 1, hops do
        if not rootPart() then finishMove() return false end
        local alpha = i / hops
        local point = start.Position + delta * alpha
        pivotRaw(CFrame.new(point, point + target.LookVector))
        RunService.Heartbeat:Wait()
    end

    pivotRaw(target)
    finishMove()
    return true
end

local function pivotTo(cframe) return travel(cframe) end

local function tpToEgg(info)
    if not info.position then return false end
    -- Land on the nest, not floating above it: less fall, less clipping.
    return travel(CFrame.new(info.position + Vector3.new(0, CFG.TPHeight, 3), info.position))
end

-- Your plot is the one whose sign carries your name.
local function resolveHome()
    local plots = Workspace:FindFirstChild("Plots")
    if not plots then return nil end
    for _, plot in ipairs(plots:GetChildren()) do
        local sign  = plot:FindFirstChild("PlotSign")
        local gui   = sign and sign:FindFirstChild("PlayerPlotSign")
        local frame = gui and gui:FindFirstChild("Frame")
        local label = frame and frame:FindFirstChild("PlayerName")
        if label and label:IsA("TextLabel") then
            local text = label.Text
            if text == LP.DisplayName or text == LP.Name
                or string.find(text, LP.Name, 1, true)
                or string.find(text, LP.DisplayName, 1, true) then
                local anchor = plot:FindFirstChild("CenterPoint") or plot:FindFirstChild("SpawnPoint")
                if anchor and anchor:IsA("BasePart") then
                    STATE.plotName = plot.Name
                    return anchor.CFrame + Vector3.new(0, 4, 0)
                end
            end
        end
    end
    return nil
end

-- Zones live under Workspace.__OBJECTS.Areas.GuardAreas as Models. The nicest
-- landing spot is a nest inside them (that is where the eggs sit); fall back to
-- the exit marker, then the bounds part, then the model pivot.
local function zoneAnchor(model)
    local nests = model:FindFirstChild("Nests")
    if nests then
        for _, nest in ipairs(nests:GetChildren()) do
            local spot = nest:FindFirstChild("EggSpotBottom")
            if spot and spot:IsA("BasePart") then
                return spot.CFrame + Vector3.new(0, 5, 0)
            end
        end
    end
    local exit = model:FindFirstChild("ClosestExitPoint")
    if exit and exit:IsA("BasePart") then return exit.CFrame + Vector3.new(0, 5, 0) end
    local bounds = model:FindFirstChild("Bounds")
    if bounds and bounds:IsA("BasePart") then
        return CFrame.new(bounds.Position + Vector3.new(0, bounds.Size.Y / 2 + 5, 0))
    end
    local ok, pivot = pcall(function() return model:GetPivot() end)
    if ok then return pivot + Vector3.new(0, 5, 0) end
    return nil
end

local function listZones()
    local out = {}
    local objects = Workspace:FindFirstChild("__OBJECTS")
    local areas = objects and objects:FindFirstChild("Areas")
    local guard = areas and areas:FindFirstChild("GuardAreas")
    if guard then
        for _, model in ipairs(guard:GetChildren()) do
            if model:IsA("Model") then
                out[#out + 1] = {name = model.Name, model = model}
            end
        end
    end
    table.sort(out, function(a, b) return a.name < b.name end)
    return out
end

local Notify -- defined with the UI

local function bank()
    if not STATE.homeCF then STATE.homeCF = resolveHome() end
    if not STATE.homeCF then
        STATE.lastMsg = "no plot found"
        return false
    end

    travel(STATE.homeCF)

    -- Hold the plot until the server claims the egg. Do NOT re-pivot every
    -- frame: that is what was fighting the physics and stuttering you around.
    local deadline = os.clock() + CFG.BankTimeout
    local hrp = rootPart()
    local anchored = false
    if CFG.HoldMode == 3 and hrp then
        hrp.Anchored = true
        anchored = true
    end

    while STATE.carrying and os.clock() < deadline do
        task.wait(0.2)
        if CFG.HoldMode == 2 then
            hrp = rootPart()
            if hrp and (hrp.Position - STATE.homeCF.Position).Magnitude > CFG.DriftStuds then
                pivotRaw(STATE.homeCF)
            end
        end
    end

    if anchored then
        hrp = rootPart()
        if hrp then hrp.Anchored = false end
    end
    return not STATE.carrying
end

local function stealOne(info)
    local record = info.record
    local origin = rootPart() and rootPart().CFrame or nil

    if CFG.SafeZones then
        local ok, req = canEscape(info.area)
        if not ok then
            STATE.lastMsg = string.format("%s guard needs %s speed, you have %s",
                info.area, short(req), short(mySpeedPower()))
            return false
        end
    end

    if not tpToEgg(info) then
        STATE.lastMsg = "no character"
        return false
    end
    RunService.Heartbeat:Wait() -- let the position replicate before asking

    local ok, reason = Net.Carry(record.Uid, slotKeyFor(record))
    if not ok then
        STATE.denied += 1
        STATE.lastMsg = "denied: " .. tostring(reason or "unknown")
        if origin then pivotTo(origin) end
        return false
    end

    STATE.carrying = true
    STATE.carryUid = record.Uid

    if CFG.AutoBank then
        if bank() then
            STATE.stolen += 1
            STATE.earned += info.value
            STATE.lastMsg = string.format("banked %s (%s/s)", info.name, short(info.value))
        else
            STATE.lastMsg = "carry ok, bank timed out"
        end
    else
        STATE.stolen += 1
        STATE.lastMsg = string.format("carrying %s", info.name)
    end
    return true
end

local function stealNearest()
    local targets = collectTargets()
    if #targets == 0 then
        STATE.lastMsg = "nothing to steal"
        return false
    end
    return stealOne(targets[1])
end

local loopThread
local function startLoop()
    if STATE.running then return end
    STATE.running = true
    STATE.homeCF = resolveHome()
    loopThread = task.spawn(function()
        while STATE.running do
            if STATE.carrying and CFG.AutoBank then bank() end
            local targets = collectTargets()
            if #targets == 0 then
                STATE.lastMsg = "waiting for spawns"
                task.wait(1)
            else
                stealOne(targets[1])
                task.wait(CFG.Delay)
            end
        end
    end)
end

local function stopLoop()
    STATE.running = false
    if loopThread then
        task.cancel(loopThread)
        loopThread = nil
    end
    STATE.lastMsg = "stopped"
end

local pages, tabButtons, activeTab = {}, {}, nil

local function makePage(name)
    local page = New("ScrollingFrame", {
        Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, BorderSizePixel = 0,
        ScrollBarThickness = 2, ScrollBarImageColor3 = T.LINE,
        CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y,
        Visible = false, Parent = PageHolder,
    })
    New("UIListLayout", {Padding = UDim.new(0, 8), SortOrder = Enum.SortOrder.LayoutOrder, Parent = page})
    pages[name] = page
    return page
end

local function selectTab(name)
    activeTab = name
    for tabName, page in pairs(pages) do page.Visible = (tabName == name) end
    for tabName, btn in pairs(tabButtons) do
        local on = tabName == name
        tw(btn, 0.18, nil, nil, {BackgroundColor3 = on and T.ACCENT or T.RAISED}):Play()
        tw(btn, 0.18, nil, nil, {TextColor3 = on and T.VOID or T.MUTED}):Play()
    end
end

local function makeTab(name, order)
    local btn = New("TextButton", {
        Size = UDim2.fromOffset(math.floor((PANEL_W - 32 - 18) / 4), 26),
        BackgroundColor3 = T.RAISED, AutoButtonColor = false, Text = name,
        TextColor3 = T.MUTED, Font = Enum.Font.GothamBold, TextSize = 11,
        LayoutOrder = order, Parent = TabBar,
    })
    Corner(btn, 8)
    Stroke(btn, T.LINE, 1, 0.6)
    btn.MouseButton1Click:Connect(function() selectTab(name) end)
    tabButtons[name] = btn
    makePage(name)
    return btn
end

    return orders[page]
end

local function makeToggle(page, label, get, set)
    local box = makeRow(page, 48)
    local lbl = rowLabel(box, label)
    local sw = New("TextButton", {
        Size = UDim2.fromOffset(48, 26), Position = UDim2.new(1, -62, 0.5, -13),
        BackgroundColor3 = T.SURFACE, AutoButtonColor = false, Text = "", Parent = box,
    })
    Corner(sw, 13)
    local swStroke = Stroke(sw, T.LINE, 1, 0.5)
    local knob = New("Frame", {
        Size = UDim2.fromOffset(18, 18), Position = UDim2.new(0, 4, 0.5, -9),
        BackgroundColor3 = T.MUTED, BorderSizePixel = 0, Parent = sw,
    })
    Corner(knob, 9)
    local function paint(on)
        tw(sw, 0.18, nil, nil, {BackgroundColor3 = on and T.ACCENT or T.SURFACE}):Play()
        tw(swStroke, 0.18, nil, nil, {Color = on and T.HIGH or T.LINE}):Play()
        TS:Create(knob, TweenInfo.new(0.26, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Position = on and UDim2.new(1, -22, 0.5, -9) or UDim2.new(0, 4, 0.5, -9),
            BackgroundColor3 = on and T.WHITE or T.MUTED,
        }):Play()
        tw(lbl, 0.18, nil, nil, {TextColor3 = on and T.TEXT or T.MUTED}):Play()
    end
    sw.MouseButton1Click:Connect(function()
        local on = not get()
        set(on); paint(on)
        Notify(label .. (on and " on" or " off"), on and T.GREEN or T.MUTED)
    end)
    paint(get())
    return paint
end

local function makeStepper(page, label, get, set, min, max, step, fmt)
    local box = makeRow(page, 48)
    rowLabel(box, label)
    local count = New("TextLabel", {
        Size = UDim2.fromOffset(52, 28), Position = UDim2.new(1, -80, 0.5, -14),
        BackgroundTransparency = 1, Text = fmt(get()), TextColor3 = T.TEXT,
        Font = Enum.Font.GothamBold, TextSize = 12, Parent = box,
    })
    local function bump(dir)
        local v = math.clamp(get() + dir * step, min, max)
        v = math.floor(v * 100 + 0.5) / 100
        set(v); count.Text = fmt(v)
    end
    local minus = New("TextButton", {
        Size = UDim2.fromOffset(28, 28), Position = UDim2.new(1, -112, 0.5, -14),
        BackgroundColor3 = T.SURFACE, AutoButtonColor = false, Text = "-",
        TextColor3 = T.HIGH, Font = Enum.Font.GothamBold, TextSize = 16, Parent = box,
    })
    Corner(minus, 7)
    local plus = New("TextButton", {
        Size = UDim2.fromOffset(28, 28), Position = UDim2.new(1, -30, 0.5, -14),
        BackgroundColor3 = T.SURFACE, AutoButtonColor = false, Text = "+",
        TextColor3 = T.HIGH, Font = Enum.Font.GothamBold, TextSize = 16, Parent = box,
    })
    Corner(plus, 7)
    minus.MouseButton1Click:Connect(function() bump(-1) end)
    plus.MouseButton1Click:Connect(function() bump(1) end)
end

local function ripple(btn, tint)
    btn.MouseButton1Down:Connect(function(x, y)
        local abs = btn.AbsolutePosition
        local size = math.max(btn.AbsoluteSize.X, btn.AbsoluteSize.Y) * 2
        local r = New("Frame", {
            Size = UDim2.fromOffset(0, 0),
            Position = UDim2.fromOffset(x - abs.X, y - abs.Y),
            AnchorPoint = Vector2.new(0.5, 0.5), BackgroundColor3 = tint or T.WHITE,
            BackgroundTransparency = 0.75, BorderSizePixel = 0, ZIndex = 10, Parent = btn,
        })
        Corner(r, 999)
        tw(r, 0.45, Enum.EasingStyle.Quad, nil, {Size = UDim2.fromOffset(size, size), BackgroundTransparency = 1}):Play()
        task.delay(0.5, function() r:Destroy() end)
    end)
end

local function makeButton(page, label, tint, onClick, height)
    local btn = New("TextButton", {
        Size = UDim2.new(1, -6, 0, height or 40), BackgroundColor3 = T.RAISED,
        AutoButtonColor = false, Text = label, TextColor3 = tint or T.HIGH,
        Font = Enum.Font.GothamBold, TextSize = 12,
        LayoutOrder = nextOrder(page), Parent = page,
    })
    Corner(btn, 12); Stroke(btn, T.LINE, 1, 0.5)
    btn.MouseEnter:Connect(function()
        tw(btn, 0.15, nil, nil, {BackgroundColor3 = T.RAISED:Lerp(tint or T.ACCENT, 0.18)}):Play()
    end)
    btn.MouseLeave:Connect(function()
        tw(btn, 0.2, nil, nil, {BackgroundColor3 = T.RAISED}):Play()
    end)
    ripple(btn, tint)
    btn.MouseButton1Click:Connect(onClick)
    return btn
end

        BackgroundTransparency = 1, BorderSizePixel = 0, Parent = NotifRoot,

        BackgroundTransparency = 1, Text = text, TextColor3 = color or T.TEXT,
        Font = Enum.Font.GothamSemibold, TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left, TextTransparency = 1, Parent = card,
    })
    tw(card, 0.2, nil, nil, {BackgroundTransparency = 0.1}):Play()
    tw(stroke, 0.2, nil, nil, {Transparency = 0.4}):Play()
    tw(lbl, 0.2, nil, nil, {TextTransparency = 0}):Play()
    task.delay(2.6, function()
        tw(card, 0.25, nil, nil, {BackgroundTransparency = 1}):Play()
        tw(stroke, 0.25, nil, nil, {Transparency = 1}):Play()
        tw(lbl, 0.25, nil, nil, {TextTransparency = 1}):Play()
        task.delay(0.3, function() card:Destroy() end)
    end)

local function mutationSummary(info)
    if #info.mutations == 0 then return "no mutations" end
    local parts = {}
    for _, m in ipairs(info.mutations) do
        parts[#parts + 1] = string.format("%s x%.3g", m.name, m.multi)
    end
    return table.concat(parts, ", ")
end

local function buildEggRow(info, index)
    local box = New("Frame", {
        Size = UDim2.new(1, 0, 0, 78), BackgroundColor3 = T.RAISED,
        LayoutOrder = index, Parent = EggListHolder,

local function rebuildEggList()
    for _, child in ipairs(EggListHolder:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end
    local targets = collectTargets()
    local shown = math.min(#targets, 40)
    local total = 0
    for _, info in ipairs(targets) do total += info.value end
    EggHeaderText.Text = string.format("%d eggs  |  %s/s on the map", #targets, short(total))
    for i = 1, shown do buildEggRow(targets[i], i) end
    if #targets > shown then
        New("TextLabel", {
            Size = UDim2.new(1, 0, 0, 20), BackgroundTransparency = 1,
            Text = string.format("... %d more, sorted by %s", #targets - shown, SORT_NAMES[CFG.SortMode]),
            TextColor3 = T.MUTED, Font = Enum.Font.Gotham, TextSize = 10,
            LayoutOrder = shown + 1, Parent = EggListHolder,
        })
    end
end

RefreshBtn.MouseButton1Click:Connect(rebuildEggList)

do -- drag
    local dragging, dragStart, startPos
    Handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = Panel.Position
        end
    end)
    UIS.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            Panel.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                                       startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

        Notify("auto steal " .. (CFG.AutoSteal and "on" or "off"), CFG.AutoSteal and T.GREEN or T.MUTED)
    elseif input.KeyCode == CFG.UIKey then
        Panel.Visible = not Panel.Visible
    end

-- Never leave a corpse noclipped, and say what killed the run.
local function watchCharacter(char)
    local hum = char:WaitForChild("Humanoid", 10)
    if not hum then return end
    hum.Died:Connect(function()
        table.clear(clipCache)
        STATE.carrying = false
        if STATE.running then
            stopLoop()
            CFG.AutoSteal = false
            paintAuto(false)
        end
        STATE.lastMsg = "died - auto stopped"
        Notify("died, auto steal stopped", T.RED)
    end)
end

    task.wait(1.5)
    table.clear(clipCache)
    STATE.homeCF = resolveHome()
    STATE.carrying = false
    watchCharacter(char)
end)
